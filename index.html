function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("AppState") || ss.insertSheet("AppState");
  var raw = sheet.getRange("A1").getValue();
  return ContentService.createTextOutput(raw ? raw : "{}").setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var body = e.postData && e.postData.contents ? e.postData.contents : "{}";
    var data = JSON.parse(body);

    var stateSheet = ss.getSheetByName("AppState") || ss.insertSheet("AppState");
    stateSheet.getRange("A1").setValue(body);
    stateSheet.getRange("B1").setValue(new Date());

    syncTableSheet(ss, "Requests", ["วันที่", "เวลา", "บ้านเลขที่", "จุดรับพัสดุ", "เบอร์โทร", "สถานะ"],
      (data.requests || []).map(function(r) {
        return [r.date, r.time, r.house, r.locationName, r.phone, r.matched ? "พบพัสดุแล้ว" : "ยังไม่มีพัสดุ"];
      })
    );

    syncTableSheet(ss, "Parcels", ["วันที่รับเข้า", "บ้านเลขที่", "เลขอ้างอิง", "ประเภทพัสดุ"],
      (data.parcels || []).map(function(p) {
        return [p.date, p.house, p.carrier, p.recipient];
      })
    );

    return ContentService.createTextOutput(JSON.stringify({ status: "ok", updated: new Date() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function syncTableSheet(ss, sheetName, headers, rows) {
  var sh = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  sh.clearContents();
  sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  if (rows && rows.length > 0) {
    sh.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}

ระบบความปลอดภัยของเว็บเบราว์เซอร์และ Sandbox ของแชทบอทจะบล็อกคำสั่งดาวน์โหลดไฟล์และสั่งพิมพ์ (`window.print()`) อัตโนมัติในหน้าพรีวิวของแชทบอท
วิธีแก้ไขเพื่อให้พิมพ์และบันทึก PDF ได้สมบูรณ์ 100%: เพียงกดที่ปุ่ม ดาวน์โหลด (ไอคอนลูกศรชี้ลงบนแถบเครื่องมือมุมขวาบนของหน้านี้) เพื่อเซฟไฟล์เว็บแอปหลัก (`.html`) ไปเปิดใช้งานบนเครื่องคอมพิวเตอร์ของคุณโดยตรง ปุ่มบันทึกเป็น PDF และฟีเจอร์ดาวน์โหลดทั้งหมดจะทำงานได้ทันทีอย่างสมบูรณ์แบบครับ!
