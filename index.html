📍 การจัดการสถานที่รับพัสดุ (Locations)

ชื่อสถานที่รับพัสดุเงื่อนไขเบอร์โทรศัพท์จัดการ

🏘️ การสร้างช่วงบ้านเลขที่อัตโนมัติ (House Generator)

ระบุคำนำหน้า (Prefix), เลขเริ่มต้น, เลขสิ้นสุด และเลขยกเว้น

🖼️ การจัดการภาพประกาศหน้าบ้าน (Announcements)

รองรับการอัปโหลดภาพ (ย่อขนาดอัตโนมัติไม่ให้เต็มความจำ) หรือวางลิงก์รูปภาพ

ลำดับภาพภาพประกาศหัวข้อประกาศจัดเรียงลำดับจัดการ

🛡️ การจัดการบัญชีและสิทธิ์ผู้ใช้หลังบ้าน (Admins)

กำหนดบัญชีผู้ใช้งานหลังบ้าน ระดับสิทธิ์ (Role) และเพิ่ม/ลบสิทธิ์แอดมิน

💬 การจัดการข้อความประกาศด่วนหน้าบ้าน (Text Announcements)

พิมพ์ข้อความประกาศหรือข่าวสารสำคัญเพื่อแสดงที่แถบประกาศด้านล่างหน้าฟอร์มคำขอหน้าบ้าน

ลำดับเนื้อหาข้อความประกาศจัดการ
ยังไม่มีข้อความประกาศในระบบ

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'getState';
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('AppState') || ss.insertSheet('AppState');
  var raw = sheet.getRange('A1').getValue();
  return ContentService.createTextOutput(raw || '{"ok":true,"empty":true}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var payload = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (payload.action === 'backupDrive') {
      var fileName = 'Parcel_Backup_' + Utilities.formatDate(new Date(), 'GMT+7', 'yyyyMMdd_HHmmss') + '.json';
      var file = DriveApp.createFile(fileName, JSON.stringify(payload.state, null, 2), MimeType.PLAIN_TEXT);
      return ContentService.createTextOutput(JSON.stringify({ ok: true, fileUrl: file.getUrl(), fileName: fileName }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    var sheet = ss.getSheetByName('AppState') || ss.insertSheet('AppState');
    sheet.getRange('A1').setValue(JSON.stringify(payload.state));
    sheet.getRange('B1').setValue(new Date());
    return ContentService.createTextOutput(JSON.stringify({ ok: true, savedAt: new Date().toISOString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
