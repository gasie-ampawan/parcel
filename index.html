📜 โค้ด Google Apps Script สำเร็จรูป (สำหรับวางใน Google Sheets -> ส่วนขยาย -> Apps Script)

วิธีติดตั้ง: เปิด Google Sheets ใหม่ -> คลิก ส่วนขยาย (Extensions) > Apps Script -> วางโค้ดด้านล่างทับทั้งหมด -> กด ทำให้ใช้งานได้ (Deploy) > รายการใหม่ (New deployment) > เว็บแอป (Web app) -> ตั้งสิทธิ์ผู้เข้าถึงเป็น ทุกคน (Anyone) -> คัดลอก URL ที่ลงท้ายด้วย /exec มาวางในช่องด้านบน 

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('AppState') || ss.insertSheet('AppState');
  var raw = sheet.getRange('A1').getValue();
  var data = raw ? JSON.parse(raw) : { ok: true, empty: true };
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (payload.action === 'backup_drive') {
      var fileName = 'Juristic_Parcel_Backup_' + Utilities.formatDate(new Date(), 'GMT+7', 'yyyyMMdd_HHmmss') + '.json';
      var file = DriveApp.createFile(fileName, JSON.stringify(payload.state, null, 2), MimeType.PLAIN_TEXT);
      return ContentService.createTextOutput(JSON.stringify({ status: 'success', fileUrl: file.getUrl(), fileName: fileName }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var stateSheet = ss.getSheetByName('AppState') || ss.insertSheet('AppState');
    stateSheet.getRange('A1').setValue(JSON.stringify(payload.state));
    stateSheet.getRange('B1').setValue(new Date());

    if (payload.state && Array.isArray(payload.state.requests)) {
      var reqSheet = ss.getSheetByName('Requests') || ss.insertSheet('Requests');
      reqSheet.clear();
      reqSheet.appendRow(['ID', 'วันที่', 'เวลา', 'บ้านเลขที่', 'จุดรับพัสดุ', 'เบอร์โทร', 'หมายเหตุ', 'สถานะ']);
      payload.state.requests.forEach(function(r) {
        reqSheet.appendRow([r.id, r.date, r.time, r.house, r.locationName, r.phone, r.note || '-', r.matched ? 'พร้อมจ่าย' : 'รอพัสดุ']);
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success', syncedAt: new Date().toISOString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
